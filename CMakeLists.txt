cmake_minimum_required(VERSION 3.24)
project(marionette VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

option(MN_BUILD_TESTS "Build unit/integration tests" ON)
option(MN_BUILD_DRIVER "Build the SteamVR (OpenVR) driver" ON)
option(MN_WITH_OPENVR_CLIENT "Build OpenVR client support (playspace calibration)" ON)
option(MN_BUILD_DASHBOARD "Build the web dashboard" ON)

if(MSVC)
    add_compile_options(/W4 /permissive- /utf-8 /EHsc)
else()
    add_compile_options(-Wall -Wextra)
endif()

include(cmake/Deps.cmake)

# --- Optional Kinect SDKs (Windows only) ---------------------------------
set(MN_HAS_KINECT_V2 OFF)
set(MN_HAS_KINECT_V1 OFF)
if(WIN32)
    if(DEFINED ENV{KINECTSDK20_DIR} AND EXISTS "$ENV{KINECTSDK20_DIR}/inc/Kinect.h")
        set(MN_HAS_KINECT_V2 ON)
        message(STATUS "Kinect SDK 2.0 found: $ENV{KINECTSDK20_DIR}")
    endif()
    if(DEFINED ENV{KINECTSDK10_DIR} AND EXISTS "$ENV{KINECTSDK10_DIR}/inc/NuiApi.h")
        set(MN_HAS_KINECT_V1 ON)
        message(STATUS "Kinect SDK 1.8 found: $ENV{KINECTSDK10_DIR}")
    endif()
endif()

add_subdirectory(core)
add_subdirectory(capture/mock)
if(MN_HAS_KINECT_V2)
    add_subdirectory(capture/kinect_v2)
endif()
if(MN_HAS_KINECT_V1)
    add_subdirectory(capture/kinect_v1)
endif()
add_subdirectory(endpoints/osc)
add_subdirectory(endpoints/openvr_bridge)
if(MN_BUILD_DASHBOARD)
    add_subdirectory(server)
endif()
if(MN_BUILD_DRIVER)
    add_subdirectory(driver/openvr)
endif()
add_subdirectory(app)

if(MN_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()
